cmake_minimum_required(VERSION 3.24)
project(BuilderGeometryWorker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MANIFOLD_TEST OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CROSS_SECTION OFF CACHE BOOL "" FORCE)
set(MANIFOLD_PAR OFF CACHE BOOL "" FORCE)
set(MANIFOLD_DOWNLOADS OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(manifold GIT_REPOSITORY https://github.com/elalish/manifold.git
  GIT_TAG ce50d78021d64507f89e8c9fc2c2e51018117857)
FetchContent_MakeAvailable(manifold)
add_executable(Builder.GeometryWorker main.cpp)
target_link_libraries(Builder.GeometryWorker PRIVATE manifold)
if(MSVC)
  target_compile_options(Builder.GeometryWorker PRIVATE /W4 /WX /fp:precise)
endif()
set_target_properties(Builder.GeometryWorker PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/bin/win-x64")
add_executable(Builder.WorkerFaultProbe EXCLUDE_FROM_ALL ../Builder.Checks/WorkerFaultProbe.cpp)
target_link_libraries(Builder.WorkerFaultProbe PRIVATE manifold)
set_target_properties(Builder.WorkerFaultProbe PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/bin/checks")
